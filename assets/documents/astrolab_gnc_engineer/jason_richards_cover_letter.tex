% Section heading used to divide the letter into its four narrative parts.
% \addvspace prevents adjacent spacing commands from accidentally stacking.
\newcommand{\LetterSection}[1]{%
    \par\addvspace{\SectionBefore}%
    {\fontsize{\SectionSize}{\SectionLineSize}\selectfont
     \textbf{\textcolor{CVBlack}{#1}}\par}%
    \addvspace{\SectionAfter}%
}

% Standard body paragraph. Change only \ParagraphSpacing above to control the
% visible gap after every paragraph that uses this command.
\newcommand{\LetterParagraph}[1]{%
    \noindent #1\par
    \addvspace{\ParagraphSpacing}%
}

% Final body paragraph. It deliberately adds no paragraph gap because
% \ClosingBefore controls the distance to "Yours sincerely".
\newcommand{\LetterFinalParagraph}[1]{%
    \noindent #1\par
}
